        // Executa quando a tela é redimensionada
        function mudouTamanho() {
            if (window.innerWidth >= 768) {
                itens.style.display = "block";
            } else {
                itens.style.display = "none";
            }
        }

        // Abre e fecha o menu no celular

            if (itens.style.display === "block") {

            } else {
                itens.style.display = "block";
            }
        }
